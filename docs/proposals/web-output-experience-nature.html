// 体験：列（core）＝ kind・region_id・duration_minutes（必須）・reservation_required・official_url
type ExperienceDetails = {
  courses?: { key: string; name: LocalizedText; description?: LocalizedText
              duration_minutes?: number; people_min?: number; people_max?: number
              age_min?: number; languages?: string[] }[]
  prices?: { items: { course?: string          // courses[].key
      per: "person" | "group" | "session"; audience?: "all" | "adult" | "child"
      age_min?: number; age_max?: number; people_min?: number; people_max?: number
      amount: number; amount_max?: number; currency: string; tax_included?: boolean }[] }
  schedules?: { months_from?: number; months_to?: number; weekdays?: number[]
                start_times?: string[]; end_time?: string; rain_policy?: LocalizedText }[]
  locations?: { kind: "meeting" | "activity"; lat: number; lng: number
                geo_precision: string; address?: LocalizedText; note?: LocalizedText }[]
  inclusions?: { kind: "included" | "excluded" | "bring"; item: LocalizedText }[]
  cautions?: { note: LocalizedText }[]
  booking_links?: { label: string; url: string; is_official?: boolean }[]
  sources?: Source[]   // 宿と同じ
}
// 自然：列（core）＝ kind・位置（必須）・region_id・protection
type NatureDetails = {
  seasons?: { subject: LocalizedText; month_from: number; month_to: number }[]
  hours?: { always_open?: boolean; open?: string; close?: string; closed_note?: LocalizedText }
  fees?: { audience?: "all" | "adult" | "child"; amount: number; amount_max?: number
           currency: string; note?: LocalizedText }[]   // 無料は amount 0
  access?: { station?: LocalizedText; mode?: "walk" | "bus" | "car"; minutes?: number
             parking?: { available: boolean; spaces?: number; paid?: boolean }
             route?: LocalizedText }
  trails?: { key: string; name: LocalizedText; distance_km?: number
             elevation_gain_m?: number; difficulty?: "easy" | "moderate" | "hard"
             duration_minutes?: number }[]
  restrictions?: { kind: "closure" | "volcanic_gas" | "other"; note?: LocalizedText
                   date_from?: string; date_to?: string }[]
  cautions?: { note: LocalizedText }[]
  sources?: Source[]
}

必須の項目

コア必須の列（NOT NULL・CHECK）details の必須（zod と quality.ts）今の決まりからの変更
体験地域・所要時間料金 1 つ以上・範囲の中の座標を持つ集合場所 1 つ以上・official_url か予約先場所を details.locations で見る。料金に per（1 名・1 グループ・1 回）が要る
自然地域・位置行き方（access）access は 1 つ（配列でなくする）

宿と同じく、1 つの値で必須なものは列（NOT NULL は欠けたデータを直してから付ける）、一覧の必須は details の中で zod と quality.ts で確かめる。旧い表は集め直しの後に消す。

画面の並び

体験

紹介文の下に「所要・予約」の 1 行 → 料金（参考：コースごとに 1 行。区分の値は出さない）→ コース → 開催（時期・曜日・時間）→ 集合場所（地図）→ 含まれるもの → 注意点 → 予約 → 出典

自然

紹介文の下に「見頃・入場」の 1 行 → 見頃 → 時間・料金 → アクセス（地図）→ 歩く道 → 規制 → 注意点 → 出典
